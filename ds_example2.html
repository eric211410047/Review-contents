雙向鏈結串列實作

程式碼

#include <stdio.h>
#include <stdlib.h>

// 定義節點結構
typedef struct Node {
    int data;
    struct Node* prev;
    struct Node* next;
} Node;

// 定義鏈結串列結構
typedef struct {
    Node* head;
    Node* tail;
    int size;
} DoublyLinkedList;

// 初始化鏈結串列
DoublyLinkedList* createList() {
    DoublyLinkedList* list = (DoublyLinkedList*)malloc(sizeof(DoublyLinkedList));
    list->head = NULL;
    list->tail = NULL;
    list->size = 0;
    return list;
}

// 在串列前端插入節點
void insertFront(DoublyLinkedList* list, int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->data = data;
    newNode->prev = NULL;
    newNode->next = list->head;
    
    if (list->head != NULL) {
        list->head->prev = newNode;
    } else {
        list->tail = newNode;
    }
    
    list->head = newNode;
    list->size++;
}

// 在串列尾端插入節點
void insertBack(DoublyLinkedList* list, int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->data = data;
    newNode->next = NULL;
    newNode->prev = list->tail;
    
    if (list->tail != NULL) {
        list->tail->next = newNode;
    } else {
        list->head = newNode;
    }
    
    list->tail = newNode;
    list->size++;
}

// 刪除串列前端的節點
void deleteFront(DoublyLinkedList* list) {
    if (list->head == NULL) return;
    
    Node* temp = list->head;
    list->head = list->head->next;
    
    if (list->head != NULL) {
        list->head->prev = NULL;
    } else {
        list->tail = NULL;
    }
    
    free(temp);
    list->size--;
}

// 刪除串列尾端的節點
void deleteBack(DoublyLinkedList* list) {
    if (list->tail == NULL) return;
    
    Node* temp = list->tail;
    list->tail = list->tail->prev;
    
    if (list->tail != NULL) {
        list->tail->next = NULL;
    } else {
        list->head = NULL;
    }
    
    free(temp);
    list->size--;
}

// 從前往後遍歷並印出串列
void printForward(DoublyLinkedList* list) {
    Node* current = list->head;
    printf("從前往後遍歷: ");
    while (current != NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");
}

// 從後往前遍歷並印出串列
void printBackward(DoublyLinkedList* list) {
    Node* current = list->tail;
    printf("從後往前遍歷: ");
    while (current != NULL) {
        printf("%d ", current->data);
        current = current->prev;
    }
    printf("\n");
}

// 釋放串列記憶體
void freeList(DoublyLinkedList* list) {
    Node* current = list->head;
    while (current != NULL) {
        Node* temp = current;
        current = current->next;
        free(temp);
    }
    free(list);
}

int main() {
    // 創建並初始化鏈結串列
    DoublyLinkedList* list = createList();
    
    // 插入一些數據
    insertFront(list, 1);
    insertBack(list, 2);
    insertFront(list, 3);
    insertBack(list, 4);
    
    // 印出串列
    printForward(list);
    printBackward(list);
    
    // 刪除節點
    deleteFront(list);
    deleteBack(list);
    
    // 再次印出串列
    printForward(list);
    printBackward(list);
    
    // 釋放記憶體
    freeList(list);
    
    return 0;
}

程式說明

這個程式實現了一個雙向鏈結串列（Doubly Linked List），具有以下特點：

  • 每個節點包含數據和兩個指針（前向和後向）
  • 支援在串列前端和尾端插入節點
  • 支援刪除串列前端和尾端的節點
  • 可以從前往後和從後往前遍歷串列
  • 包含完整的記憶體管理（分配和釋放）
