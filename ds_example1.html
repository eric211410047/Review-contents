資料結構範例：256x256乘法表

程式碼

#include <stdio.h> #include <limits.h> #include <stdlib.h> #define MATRIX_SIZE 257 // 0-256需要257個元素 typedef unsigned char U8; typedef unsigned int U32; void dump_matrix(int** matrix); void store_matrix(int** matrix); int main() { //$ Part 1: U32 u32Idx = 0; U32** matrix = (U32**)malloc(sizeof(U32*)*MATRIX_SIZE); for (u32Idx = 0; u32Idx < MATRIX_SIZE; u32Idx++) matrix[u32Idx] = (U32*)malloc(sizeof(U32)*MATRIX_SIZE); //$ Part 2: store_matrix(matrix); dump_matrix(matrix); // 釋放記憶體 for (u32Idx = 0; u32Idx < MATRIX_SIZE; u32Idx++) free(matrix[u32Idx]); free(matrix); return 0; } void dump_matrix(int** matrix) { U32 u32Idx = 0; // 只輸出平方數 (i=j 的情況) for (u32Idx = 0; u32Idx <= 256; u32Idx++) { printf("%dX%d=%d", u32Idx, u32Idx, matrix[u32Idx][u32Idx]); // 每行顯示4個結果 if (u32Idx < 256) { if ((u32Idx + 1) % 4 == 0) printf("\n"); else printf(" "); } } printf("\n"); } void store_matrix(int** matrix) { U32 u32Idx1 = 0; U32 u32Idx2 = 0; for (u32Idx1 = 0; u32Idx1 < MATRIX_SIZE; u32Idx1++) { for (u32Idx2 = 0; u32Idx2 < MATRIX_SIZE; u32Idx2++) { matrix[u32Idx1][u32Idx2] = u32Idx1 * u32Idx2; } } }

程式碼解釋

這個程式實現了一個256x256的乘法表，使用了二維動態陣列作為資料結構。程式主要分為三個部分：

  1. 記憶體分配：使用malloc動態分配一個257x257的二維陣列（包含0-256的所有數字）
  2. 資料填充：將乘法表的結果存入陣列中
  3. 資料輸出：只輸出平方數（即i=j的情況），每行顯示4個結果

最後，程式會釋放所有分配的記憶體，避免記憶體洩漏。
